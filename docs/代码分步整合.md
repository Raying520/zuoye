```java
package com.campus.activity.entity;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.EnumType;
import jakarta.persistence.Enumerated;
import jakarta.persistence.GeneratedValue;
import jakarta.persistence.GenerationType;
import jakarta.persistence.Id;
import jakarta.persistence.Table;

/**
 * 用户实体（对应数据表 app_user，逻辑名"用户表"）
 * 角色：STUDENT 学生 / TEACHER 教师
 */
@Entity
@Table(name = "app_user")
public class User {

    /** 用户角色 */
    public enum Role { STUDENT, TEACHER }

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    /** 登录账号，唯一 */
    @Column(nullable = false, unique = true, length = 50)
    private String username;

    /** BCrypt 哈希后的密码，不存明文 */
    @Column(nullable = false)
    private String password;

    /** 姓名 */
    @Column(nullable = false, length = 50)
    private String name;

    /** 角色：STUDENT / TEACHER */
    @Enumerated(EnumType.STRING)
    @Column(nullable = false, length = 20)
    private Role role;

    public Long getId() { return id; }

    public void setId(Long id) { this.id = id; }

    public String getUsername() { return username; }

    public void setUsername(String username) { this.username = username; }

    public String getPassword() { return password; }

    public void setPassword(String password) { this.password = password; }

    public String getName() { return name; }

    public void setName(String name) { this.name = name; }

    public Role getRole() { return role; }

    public void setRole(Role role) { this.role = role; }
}
```

### `main/java/com/campus/activity/service/UserService.java`

```java
package com.campus.activity.service;

import com.campus.activity.dto.LoginForm;
import com.campus.activity.dto.RegisterForm;
import com.campus.activity.dto.UserSession;
import com.campus.activity.entity.User;
import com.campus.activity.repository.UserRepository;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.stereotype.Service;

import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

/**
 * 用户服务：业务规则集中层
 * 注册流程（对应软件设计 3.1）：
 * 填写信息 → 选择角色 → 账号唯一性检查 → 密码 BCrypt 哈希 → 入库
 */
@Service
public class UserService {

    private final UserRepository userRepository;
    private final BCryptPasswordEncoder passwordEncoder = new BCryptPasswordEncoder();

    public UserService(UserRepository userRepository) {
        this.userRepository = userRepository;
    }

    /**
     * 登录校验（FR2）
     * 按账号查询用户，BCrypt 比对密码。
     *
     * @return 登录会话信息
     * @throws IllegalArgumentException 账号不存在或密码错误
     */
    public UserSession login(LoginForm form) {
        User user = userRepository.findByUsername(form.getUsername())
                .orElseThrow(() -> new IllegalArgumentException("账号或密码错误"));
        if (!passwordEncoder.matches(form.getPassword(), user.getPassword())) {
            throw new IllegalArgumentException("账号或密码错误");
        }
        return UserSession.fromUser(user);
    }

    /**
     * 注册新用户（FR1）
     *
     * @throws IllegalArgumentException 账号已存在或两次密码不一致
     */
    public void register(RegisterForm form) {
        if (!form.getPassword().equals(form.getConfirmPassword())) {
            throw new IllegalArgumentException("两次输入的密码不一致");
        }
        if (userRepository.existsByUsername(form.getUsername())) {
            throw new IllegalArgumentException("该账号已被注册");
        }
        User user = new User();
        user.setUsername(form.getUsername());
        // 密码不明文入库（安全约束）
        user.setPassword(passwordEncoder.encode(form.getPassword()));
        user.setName(form.getName());
        user.setRole(form.getRole());
        userRepository.save(user);
    }

    /** 按 ID 查询用户（FR6 查看报名名单用） */
    public User findById(Long id) {
        return userRepository.findById(id).orElse(null);
    }

    /** 批量查询用户 ID 到姓名的映射（FR6 查看报名名单用） */
    public Map<Long, String> findNamesByIds(List<Long> ids) {
        return userRepository.findAllById(ids).stream()
                .collect(Collectors.toMap(User::getId, User::getName));
    }
}
```

```java
package com.campus.activity.controller;

import com.campus.activity.dto.LoginForm;
import com.campus.activity.dto.RegisterForm;
import com.campus.activity.dto.UserSession;
import com.campus.activity.interceptor.LoginInterceptor;
import com.campus.activity.service.UserService;
import jakarta.servlet.http.HttpSession;
import jakarta.validation.Valid;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.validation.BindingResult;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.servlet.mvc.support.RedirectAttributes;

/**
 * 用户控制器：注册（FR1）
 */
@Controller
public class UserController {

    private final UserService userService;

    public UserController(UserService userService) {
        this.userService = userService;
    }

    /** 注册页面 */
    @GetMapping("/register")
    public String registerPage(RegisterForm form) {
        return "register";
    }

    /** 提交注册 */
    @PostMapping("/register")
    public String register(@Valid RegisterForm form,
                           BindingResult bindingResult,
                           Model model,
                           RedirectAttributes redirectAttributes) {
        // 1. 表单校验失败：回到注册页显示字段错误
        if (bindingResult.hasErrors()) {
            return "register";
        }
        // 2. 业务校验（账号唯一、两次密码一致）
        try {
            userService.register(form);
        } catch (IllegalArgumentException e) {
            model.addAttribute("bizError", e.getMessage());
            return "register";
        }
        // 3. 注册成功，跳转到登录页
        redirectAttributes.addFlashAttribute("success", "注册成功，请登录");
        return "redirect:/login";
    }

    /** 登录页面 */
    @GetMapping("/login")
    public String loginPage(LoginForm loginForm) {
        return "login";
    }

    /** 提交登录（FR2） */
    @PostMapping("/login")
    public String login(@Valid LoginForm loginForm,
                        BindingResult bindingResult,
                        Model model,
                        HttpSession session) {
        // 1. 表单校验失败：回到登录页显示字段错误
        if (bindingResult.hasErrors()) {
            return "login";
        }
        // 2. 业务校验（账号存在且密码匹配）
        UserSession userSession;
        try {
            userSession = userService.login(loginForm);
        } catch (IllegalArgumentException e) {
            model.addAttribute("bizError", e.getMessage());
            return "login";
        }
        // 3. 写入 Session，完成角色分流
        session.setAttribute(LoginInterceptor.SESSION_KEY, userSession);
        if (userSession.isTeacher()) {
            return "redirect:/teacher/home";
        }
        return "redirect:/student/home";
    }

    /** 登出 */
    @GetMapping("/logout")
    public String logout(HttpSession session) {
        session.invalidate();
        return "redirect:/";
    }
}
```

```java
package com.campus.activity.dto;
```

```java
import java.io.Serializable;

public class UserSession implements Serializable {

    public String getUsername() { return username; }

    public void setUsername(String username) { this.username = username; }

    public boolean isStudent() { return role == User.Role.STUDENT; }
```

### `main/java/com/campus/activity/interceptor/LoginInterceptor.java`

```java
package com.campus.activity.interceptor;

import com.campus.activity.dto.UserSession;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import jakarta.servlet.http.HttpSession;
import org.springframework.stereotype.Component;
import org.springframework.web.servlet.HandlerInterceptor;

/**
 * 登录拦截器（对应软件设计 3.4：登录拦截器校验会话）
 * 未登录用户访问受保护路径时，跳转登录页。
 */
@Component
public class LoginInterceptor implements HandlerInterceptor {

    public static final String SESSION_KEY = "currentUser";

    @Override
    public boolean preHandle(HttpServletRequest request,
                             HttpServletResponse response,
                             Object handler) throws Exception {
        HttpSession session = request.getSession(false);
        UserSession user = session == null ? null : (UserSession) session.getAttribute(SESSION_KEY);
        if (user == null) {
            // 保存当前请求地址，登录成功后可根据需要跳转回来（V1.0 先统一跳首页）
            response.sendRedirect("/login");
            return false;
        }
```

```java
package com.campus.activity.service;

import com.campus.activity.dto.ActivityForm;
import com.campus.activity.dto.ActivityListItem;
import com.campus.activity.dto.UserSession;
import com.campus.activity.entity.Activity;
import com.campus.activity.repository.ActivityRepository;
import com.campus.activity.repository.EnrollmentRepository;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.ArrayList;
import java.util.List;

/**
 * 活动业务服务（FR3、FR4、FR7 的业务规则实现）
 */
@Service
public class ActivityService {

    private final ActivityRepository activityRepository;
    private final EnrollmentRepository enrollmentRepository;

    public ActivityService(ActivityRepository activityRepository,
                           EnrollmentRepository enrollmentRepository) {
        this.activityRepository = activityRepository;
        this.enrollmentRepository = enrollmentRepository;
    }

    /** 教师发布活动（FR3） */
    @Transactional
    public Activity publish(ActivityForm form, UserSession teacher) {
        if (!teacher.isTeacher()) {
            throw new IllegalArgumentException("只有教师可以发布活动");
        }
        Activity activity = new Activity();
        activity.setTitle(form.getTitle().trim());
        activity.setDescription(form.getDescription());
        activity.setActivityTime(form.getActivityTime());
        activity.setPlace(form.getPlace().trim());
        activity.setCapacity(form.getCapacity());
        activity.setStatus(Activity.Status.OPEN);
        activity.setTeacherId(teacher.getId());
        return activityRepository.save(activity);
    }

    /** 查询所有活动，按活动时间倒序（FR4） */
    public List<Activity> findAll() {
        return activityRepository.findAllByOrderByActivityTimeDesc();
    }

    /** 查询所有活动并附带当前报名人数（FR4 列表展示） */
    public List<ActivityListItem> findAllWithEnrollmentCount() {
        List<Activity> activities = activityRepository.findAllByOrderByActivityTimeDesc();
        List<ActivityListItem> result = new ArrayList<>(activities.size());
        for (Activity activity : activities) {
            long count = enrollmentRepository.countByActivityId(activity.getId());
            result.add(new ActivityListItem(activity, count));
        }
        return result;
    }

    /** 教师查询自己发布的活动（FR7） */
    public List<Activity> findByTeacher(Long teacherId) {
        return activityRepository.findByTeacherIdOrderByActivityTimeDesc(teacherId);
    }

    /** 教师查询自己发布的活动并附带报名人数（FR7 管理列表） */
    public List<ActivityListItem> findByTeacherWithEnrollmentCount(Long teacherId) {
        List<Activity> activities = activityRepository.findByTeacherIdOrderByActivityTimeDesc(teacherId);
        List<ActivityListItem> result = new ArrayList<>(activities.size());
        for (Activity activity : activities) {
            long count = enrollmentRepository.countByActivityId(activity.getId());
            result.add(new ActivityListItem(activity, count));
        }
        return result;
    }

    /** 按 ID 查询活动 */
    public Activity findById(Long id) {
        return activityRepository.findById(id).orElse(null);
    }

    /** 教师取消活动（FR7 + BR5） */
    @Transactional
    public void cancel(Long activityId, UserSession teacher) {
        Activity activity = activityRepository.findById(activityId)
                .orElseThrow(() -> new IllegalArgumentException("活动不存在"));
        if (!activity.getTeacherId().equals(teacher.getId())) {
            throw new IllegalArgumentException("只能管理自己发布的活动");
        }
        if (activity.getStatus() != Activity.Status.OPEN) {
            throw new IllegalArgumentException("活动当前状态不可取消");
        }
        activity.setStatus(Activity.Status.CANCELLED);
        activityRepository.save(activity);
    }

    /** 教师结束活动（FR7 + BR5） */
    @Transactional
    public void close(Long activityId, UserSession teacher) {
        Activity activity = activityRepository.findById(activityId)
                .orElseThrow(() -> new IllegalArgumentException("活动不存在"));
        if (!activity.getTeacherId().equals(teacher.getId())) {
            throw new IllegalArgumentException("只能管理自己发布的活动");
        }
        if (activity.getStatus() != Activity.Status.OPEN) {
            throw new IllegalArgumentException("活动当前状态不可结束");
        }
        activity.setStatus(Activity.Status.CLOSED);
        activityRepository.save(activity);
    }
}
```

```java
import jakarta.persistence.*;

/**

 */
@Entity
@Table(name = "enrollment",
       uniqueConstraints = @UniqueConstraint(
           name = "uk_enrollment_activity_user",
           columnNames = {"activity_id", "user_id"}
       ))
@EntityListeners(AuditingEntityListener.class)
```

```java
package com.campus.activity.service;

import com.campus.activity.dto.UserSession;
import com.campus.activity.entity.Activity;
import com.campus.activity.entity.Enrollment;
import com.campus.activity.repository.ActivityRepository;
import com.campus.activity.repository.EnrollmentRepository;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

/**
 * 报名业务服务（FR5 报名/取消报名、FR6 名单查看）
 * 核心规则：BR2 状态校验、BR3 容量校验、BR4 重复报名校验
 */
@Service
public class EnrollmentService {

    private final EnrollmentRepository enrollmentRepository;
    private final ActivityRepository activityRepository;

    public EnrollmentService(EnrollmentRepository enrollmentRepository,
                             ActivityRepository activityRepository) {
        this.enrollmentRepository = enrollmentRepository;
        this.activityRepository = activityRepository;
    }

    /** 学生报名（FR5）：三重校验后写入记录 */
    @Transactional
    public void enroll(Long activityId, UserSession student) {
        if (!student.isStudent()) {
            throw new IllegalArgumentException("只有学生可以报名活动");
        }
        Activity activity = activityRepository.findById(activityId)
                .orElseThrow(() -> new IllegalArgumentException("活动不存在"));

        // BR2：活动状态 = 报名中
        if (activity.getStatus() != Activity.Status.OPEN) {
            throw new IllegalArgumentException("该活动当前不可报名");
        }

        // BR3：报名人数 < 容量上限
        long enrolledCount = enrollmentRepository.countByActivityId(activityId);
        if (enrolledCount >= activity.getCapacity()) {
            throw new IllegalArgumentException("活动名额已满");
        }

        // BR4：不可重复报名（数据库唯一约束兜底）
        if (enrollmentRepository.existsByActivityIdAndUserId(activityId, student.getId())) {
            throw new IllegalArgumentException("你已经报名过该活动");
        }

        Enrollment enrollment = new Enrollment();
        enrollment.setActivityId(activityId);
        enrollment.setUserId(student.getId());
        enrollmentRepository.save(enrollment);
    }

    /** 学生取消报名（FR5）：仅活动报名中时可取消 */
    @Transactional
    public void cancel(Long activityId, UserSession student) {
        if (!student.isStudent()) {
            throw new IllegalArgumentException("只有学生可以取消报名");
        }
        Activity activity = activityRepository.findById(activityId)
                .orElseThrow(() -> new IllegalArgumentException("活动不存在"));
        if (activity.getStatus() != Activity.Status.OPEN) {
            throw new IllegalArgumentException("活动已结束或已取消，无法变更报名");
        }
        Enrollment enrollment = enrollmentRepository.findByActivityIdAndUserId(activityId, student.getId())
                .orElseThrow(() -> new IllegalArgumentException("你尚未报名该活动"));
        enrollmentRepository.delete(enrollment);
    }

    /** 查询某活动当前报名人数（BR3） */
    public long countByActivity(Long activityId) {
        return enrollmentRepository.countByActivityId(activityId);
    }

    /** 查询学生是否已报名某活动（BR4） */
    public boolean isEnrolled(Long activityId, Long userId) {
        return enrollmentRepository.existsByActivityIdAndUserId(activityId, userId);
    }

    /** 查询学生的所有报名记录（FR6） */
    public List<Enrollment> findByUser(Long userId) {
        return enrollmentRepository.findByUserIdOrderByCreateTimeDesc(userId);
    }

    /** 查询活动的所有报名记录（FR6） */
    public List<Enrollment> findByActivity(Long activityId) {
        return enrollmentRepository.findByActivityIdOrderByCreateTimeDesc(activityId);
    }
}
```

```java
        if (bindingResult.hasErrors()) {
            return "login";

        session.setAttribute(LoginInterceptor.SESSION_KEY, userSession);
```

```java
            redirectAttributes.addFlashAttribute("bizError", e.getMessage());
        }
        return "redirect:/activities/manage";
    }

    /** 教师：结束活动（FR7 + BR5） */
    @PostMapping("/{id}/close")
    public String close(@PathVariable Long id,
                        HttpSession session,
                        RedirectAttributes redirectAttributes) {
        UserSession user = currentUser(session);
        if (user == null || !user.isTeacher()) {
            redirectAttributes.addFlashAttribute("bizError", "只有教师可以管理活动");
            return "redirect:/login";
        }
        try {
            activityService.close(id, user);
            redirectAttributes.addFlashAttribute("success", "活动已结束");
        } catch (IllegalArgumentException e) {
            redirectAttributes.addFlashAttribute("bizError", e.getMessage());
        }
        return "redirect:/activities/manage";
    }

    /** 教师：查看活动报名名单（FR6） */
    @GetMapping("/{id}/enrollments")
    public String enrollments(@PathVariable Long id,
                              HttpSession session,
                              Model model) {
        UserSession user = currentUser(session);
        var activity = activityService.findById(id);
        if (activity == null || user == null || !user.isTeacher()
                || !activity.getTeacherId().equals(user.getId())) {
            model.addAttribute("bizError", "无权查看该活动的报名名单");
            return "login";
        }
        var enrollmentList = enrollmentService.findByActivity(id);
        var names = userService.findNamesByIds(
                enrollmentList.stream().map(e -> e.getUserId()).toList());
        model.addAttribute("activity", activity);
        model.addAttribute("enrollments", enrollmentList);
        model.addAttribute("names", names);
        return "activity/enrollments";
    }

    private UserSession currentUser(HttpSession session) {
        return (UserSession) session.getAttribute(LoginInterceptor.SESSION_KEY);
    }
```
